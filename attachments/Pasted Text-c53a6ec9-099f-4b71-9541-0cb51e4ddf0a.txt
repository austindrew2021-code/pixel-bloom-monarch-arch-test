🛠️ Step-by-Step Setup Guide: What to Do
Step 1: Set Up Your Supabase Tables
Log into your Supabase Dashboard, open the SQL Editor, and run this script to create your data infrastructure, user profiles, and priority job queue:
sql
-- 1. Create a profiles table to track subscription tiers CREATE TABLE profiles ( id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY, is_premium BOOLEAN DEFAULT FALSE, tokens_remaining INT DEFAULT 3, updated_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) ); -- 2. Create the Recipe Queue Table to manage traffic spikes CREATE TABLE recipe_queue ( id BIGSERIAL PRIMARY KEY, user_id UUID REFERENCES auth.users ON DELETE CASCADE, prompt TEXT NOT NULL, is_weekly_plan BOOLEAN DEFAULT FALSE, status TEXT DEFAULT 'pending', -- pending, processing, completed, failed recipe_output TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) ); -- 3. Create a Recipe Cache Table to store repetitive generations for $0 CREATE TABLE recipe_cache ( id BIGSERIAL PRIMARY KEY, clean_prompt TEXT UNIQUE NOT NULL, recipe_text TEXT NOT NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) ); -- Enable Realtime so the mobile app can listen live for completed recipes ALTER PUBLICATION supabase_realtime ADD TABLE recipe_queue;
